171:   const entries = Object.entries(value);
172:   const kept = entries.slice(0, MAX_CONTEXT_JSON_OBJECT_KEYS);
173:   const result: Array<readonly [string, unknown]> = [];
174:   let omitted = entries.length - kept.length;
175:   let budgetExhausted = false;
176:   for (const [key, entry] of kept) {
177:     if (budget.remaining <= 0) {
178:       omitted += 1;
179:       budgetExhausted = true;
180:       continue;
181:     }
182:     // Keys are channel-controlled text too: cap them like string values and
183:     // charge them to the budget so key fan-out cannot bypass the total.
184:     const safeKey = neutralizeMarkdownFences(truncateContextJsonString(key));
185:     // Reserve the nested value against a scratch budget: the recursion debits
186:     // as it works, and the property is committed to the shared budget only
187:     // when accepted whole. Debiting the shared budget during recursion and
188:     // then again for the retained value would charge nested containers twice
189:     // and drop properties that actually fit.
190:     const scratch: ContextJsonBudget = { remaining: budget.remaining };
191:     const sanitized = sanitizeContextJsonValue(entry, scratch, depth + 1);
192:     // Reserve the property's serialized size before appending: JSON.stringify
193:     // emits the key with quotes and escapes (a key of N raw chars can emit ~2N
194:     // chars), so charging raw key length would let escaped keys push the block
195:     // to nearly twice the budget while the counter stays under it.
196:     const propertySize = serializedLength(safeKey) + serializedLength(sanitized);
197:     if (propertySize > budget.remaining) {
198:       omitted += 1;
199:       budgetExhausted = true;
200:       continue;
201:     }
202:     budget.remaining -= propertySize;
203:     result.push([safeKey, sanitized] as const);
204:   }
205:   // Truncation flag mirrors the sibling `history_truncated: true` convention.
206:   if (budgetExhausted) {
207:     result.push([BUDGET_TRUNCATION_MARKER, true] as const);
208:   } else if (omitted > 0) {
209:     result.push([`…[truncated: ${omitted} more ${omitted === 1 ? "key" : "keys"}]`, true] as const);
210:   }
211:   return Object.fromEntries(result);
